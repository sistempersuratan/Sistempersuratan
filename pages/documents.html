import { db, auth } from '../js/firebase-config.js'; import { collection, getDocs, doc, getDoc } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js"; import { onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js"; // 1. Cek Sesi Login & Ambil Profil User dari Firestore onAuthStateChanged(auth, async (user) => { if (user) { try { if (!db) { console.error("Database belum terinisialisasi dengan benar."); return; } const userDocRef = doc(db, "users", user.uid); const userDoc = await getDoc(userDocRef); if (userDoc.exists()) { const userData = userDoc.data(); // Tampilkan nama dan role di sidebar const userNameEl = document.getElementById("userName"); if (userNameEl) userNameEl.textContent = userData.username || user.email; const roleBadge = document.getElementById("userRoleBadge"); if (roleBadge) { if (userData.role === "admin") { roleBadge.textContent = "Admin"; // Tampilkan elemen khusus admin document.querySelectorAll(".admin-only").forEach(el => { el.style.display = "inline-block"; }); } else { roleBadge.textContent = "User"; } } // Inisial avatar const userInitialEl = document.getElementById("userInitial"); if (userInitialEl) { userInitialEl.textContent = (userData.username || user.email).charAt(0).toUpperCase(); } // Muat data tabel dokumen loadDocumentsTable(user.uid, userData.role); } else { console.error("Dokumen profil user tidak ditemukan di koleksi users!"); } } catch (error) { console.error("Gagal memuat profil user:", error); } } else { // Jika belum login, arahkan kembali ke halaman utama window.location.href = "../index.html"; } }); // 2. Fungsi Tombol Logout document.querySelectorAll(".js-logout").forEach(btn => { btn.addEventListener("click", () => { signOut(auth).then(() => { window.location.href = "../index.html"; }).catch((error) => { console.error("Gagal keluar:", error); }); }); }); // 3. Fungsi Render Tabel Dokumen (Disesuaikan dengan HTML kamu) async function loadDocumentsTable(uid, role) { const tbody = document.getElementById("documentsTableBody"); if (!tbody) return; try { tbody.innerHTML = `Memuat data...`; const querySnapshot = await getDocs(collection(db, "documents")); tbody.innerHTML = ""; if (querySnapshot.empty) { tbody.innerHTML = `Belum ada dokumen.`; return; } querySnapshot.forEach((docSnap) => { const data = docSnap.data(); // Jika bukan admin, hanya tampilkan dokumen miliknya sendiri (jika field userId ada) if (role !== "admin" && data.userId && data.userId !== uid) return; const tr = document.createElement("tr"); // Susunan kolom harus klop dengan thead di HTML: // 1. Nomor Surat, 2. Template, 3. Tanggal, 4. Dibuat Oleh (Admin-only), 5. Status, 6. Aksi tr.innerHTML = ` ${data.nomor_surat || '-'} ${data.template_name || 'Invoice'} ${data.tanggal_dibuat || '-'} ${data.creator_name || data.userEmail || '-'} ${data.status || 'Draft'} `; tbody.appendChild(tr); }); // Event listener untuk tombol "Lihat" detail dokumen document.querySelectorAll(".view-doc-btn").forEach(btn => { btn.addEventListener("click", (e) => { const docId = e.target.getAttribute("data-id"); alert("Melihat detail dokumen ID: " + docId); // Buka panel detail atau muat data detail di sini jika diperlukan const detailPanel = document.getElementById("detailPanel"); if (detailPanel) detailPanel.style.display = "block"; }); }); } catch (error) { console.error("Gagal mengambil data dokumen:", error); tbody.innerHTML = `Gagal memuat data dari database.`; } }
